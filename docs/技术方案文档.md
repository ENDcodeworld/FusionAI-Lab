```python
# src/data_pipeline/pipeline.py

from pyspark.sql import SparkSession
from pyspark.sql.functions import *

class FusionDataPipeline:
    """聚变数据处理管道"""
    
    def __init__(self):
        self.spark = SparkSession.builder \
            .appName("FusionAI-Lab") \
            .getOrCreate()
    
    def process_papers(self, raw_papers: List[Dict]) -> DataFrame:
        """处理论文数据"""
        df = self.spark.createDataFrame(raw_papers)
        
        # 数据清洗
        df_clean = df.dropna(subset=['title', 'abstract'])
        
        # 特征提取
        df_features = df_clean \
            .withColumn('word_count', length(col('abstract'))) \
            .withColumn('pub_year', year(col('publish_date'))) \
            .withColumn('citation_count', col('citations').cast('int'))
        
        return df_features
    
    def process_funding(self, raw_funding: List[Dict]) -> DataFrame:
        """处理融资数据"""
```

```javascript
// src/frontend/components/FusionVisualizer.jsx

import React, { useEffect, useRef } from 'react';
import * as THREE from 'three';

const FusionVisualizer = ({ data }) => {
  const containerRef = useRef(null);
  
  useEffect(() => {
    // 创建 3D 托卡马克模型
    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(75, 
      window.innerWidth / window.innerHeight, 0.1, 1000);
    const renderer = new THREE.WebGLRenderer({ antialias: true });
    
    // 创建环形真空室
    const torusGeometry = new THREE.TorusGeometry(10, 3, 16, 100);
    const torusMaterial = new THREE.MeshPhongMaterial({ 
      color: 0x00aaff,
      transparent: true,
      opacity: 0.7
    });
    const torus = new THREE.Mesh(torusGeometry, torusMaterial);
    scene.add(torus);
    
    // 创建等离子体
    const plasmaGeometry = new THREE.TorusGeometry(8, 2, 16, 100);
    const plasmaMaterial = new THREE.MeshBasicMaterial({ 
      color: 0xff6600,
      transparent: true,
      opacity: 0.8
    });
    const plasma = new THREE.Mesh(plasmaGeometry, plasmaMaterial);
    scene.add(plasma);
    
    // 动画循环
    const animate = () => {
      requestAnimationFrame(animate);
      plasma.rotation.z += 0.01;
      renderer.render(scene, camera);
    };
    animate();
    
    return () => {
      renderer.dispose();
    };
  }, [data]);
  
  return <div ref={containerRef} style={{ width: '100%', height: '600px' }} />;
};

export default FusionVisualizer;
```

## 三、数据库设计

### 3.1 核心表结构

```sql
-- 公司表
CREATE TABLE companies (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    country VARCHAR(100),
    founded_year INTEGER,
    technology_type VARCHAR(100),
    website VARCHAR(255),
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 融资轮次表
CREATE TABLE funding_rounds (
    id SERIAL PRIMARY KEY,
    company_id INTEGER REFERENCES companies(id),
    round_type VARCHAR(50), -- Seed, A, B, C, etc.
    amount_usd BIGINT,
    announced_date DATE,
    lead_investor VARCHAR(255),
    investors JSONB,
    post_money_valuation BIGINT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 论文表
CREATE TABLE papers (
    id SERIAL PRIMARY KEY,
    title VARCHAR(1000) NOT NULL,
    abstract TEXT,
    authors JSONB,
    journal VARCHAR(255),
    publish_date DATE,
    doi VARCHAR(100),
    arxiv_id VARCHAR(50),
    citations INTEGER DEFAULT 0,
    pdf_url VARCHAR(500),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 实验数据表
CREATE TABLE experiment_data (
    id SERIAL PRIMARY KEY,
    facility VARCHAR(100), -- ITER, JET, EAST, etc.
    experiment_id VARCHAR(100),
    shot_number INTEGER,
    timestamp TIMESTAMP,
    parameters JSONB, -- 等离子体参数
    results JSONB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 用户表
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255),
    role VARCHAR(50) DEFAULT 'user', -- user, premium, admin
    subscription_type VARCHAR(50), -- free, basic, premium
    subscription_expires TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 报告表
CREATE TABLE reports (
    id SERIAL PRIMARY KEY,
    title VARCHAR(500) NOT NULL,
    type VARCHAR(100), -- annual, quarterly, company_analysis
    price_cny INTEGER,
    content_path VARCHAR(500),
    published_date DATE,
    author_id INTEGER REFERENCES users(id),
    download_count INTEGER DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 创建索引
CREATE INDEX idx_companies_tech ON companies(technology_type);
CREATE INDEX idx_funding_company ON funding_rounds(company_id);
CREATE INDEX idx_funding_date ON funding_rounds(announced_date);
CREATE INDEX idx_papers_date ON papers(publish_date);
CREATE INDEX idx_papers_citations ON papers(citations);
CREATE INDEX idx_experiment_facility ON experiment_data(facility);
CREATE INDEX idx_users_email ON users(email);
```
